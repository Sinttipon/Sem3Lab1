cmake_minimum_required(VERSION 3.14)
project(Lab1 CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()

include(FetchContent)
FetchContent_Declare(
    googletest
    URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.zip
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

add_library(lab1 INTERFACE)
target_include_directories(lab1 INTERFACE include)

add_executable(main src/main.cpp)
target_link_libraries(main PRIVATE lab1)

add_executable(benchmark benchmark/benchmark.cpp)
target_link_libraries(benchmark PRIVATE lab1)

enable_testing()
add_executable(tests tests/gtest.cpp)
target_link_libraries(tests PRIVATE lab1 GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(tests)
